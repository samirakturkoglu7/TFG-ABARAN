import torch
import torch.nn as nn
import torch.optim as optim
import time
import psutil
import os

DEVICE = torch.device("cuda")

# Red neuronal
class MLP(nn.Module):
    def __init__(self):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(1024, 2048),
            nn.ReLU(),
            nn.BatchNorm1d(2048),
            nn.Linear(2048, 1024),
            nn.ReLU(),
            nn.BatchNorm1d(1024),
            nn.Linear(1024, 10)
        )
    def forward(self, x):
        return self.net(x)

# Dataset sintético
X = torch.randn(5000, 1024).to(DEVICE)
y = torch.randint(0, 10, (5000,)).to(DEVICE)

model = MLP().to(DEVICE)
optimizer = optim.Adam(model.parameters(), lr=1e-3)
criterion = nn.CrossEntropyLoss()

# ── ENTRENAMIENTO ──
print("Entrenando...")
torch.cuda.synchronize()
t0 = time.perf_counter()

for epoch in range(5):
    perm = torch.randperm(5000)
    total_loss = 0
    for i in range(0, 5000, 256):
        idx = perm[i:i+256]
        optimizer.zero_grad()
        loss = criterion(model(X[idx]), y[idx])
        loss.backward()
        optimizer.step()
        total_loss += loss.item()
    print(f"  Epoch {epoch+1}/5 loss: {total_loss:.3f}")

torch.cuda.synchronize()
t_train = (time.perf_counter() - t0) * 1000
peak_vram = torch.cuda.max_memory_allocated() / 1024**2
ram = psutil.Process().memory_info().rss / 1024**2

print(f"\n── Resultados entrenamiento ──")
print(f"Tiempo:      {t_train:.0f} ms")
print(f"VRAM pico:   {peak_vram:.1f} MB")
print(f"RAM proceso: {ram:.1f} MB")

# ── INFERENCIA ──
print("\nMidiendo inferencia (100 runs, batch=1)...")
model.eval()
x_sample = torch.randn(1, 1024).to(DEVICE)
latencias = []

with torch.no_grad():
    for _ in range(10):  # warmup
        _ = model(x_sample)

    for _ in range(100):
        torch.cuda.synchronize()
        t0 = time.perf_counter()
        _ = model(x_sample)
        torch.cuda.synchronize()
        latencias.append((time.perf_counter() - t0) * 1000)

import statistics
print(f"\n── Resultados inferencia ──")
print(f"Latencia media: {statistics.mean(latencias):.3f} ms")
print(f"Latencia std:   {statistics.stdev(latencias):.3f} ms")
print(f"Latencia min:   {min(latencias):.3f} ms")
print(f"Latencia max:   {max(latencias):.3f} ms")

# ── POTENCIA ──
ina_base = "/sys/bus/i2c/drivers/ina3221/1-0040/hwmon"
if os.path.exists(ina_base):
    for hwmon in os.listdir(ina_base):
        p = f"{ina_base}/{hwmon}/power1_input"
        if os.path.exists(p):
            watts = int(open(p).read().strip()) / 1e6
            print(f"Potencia actual: {watts:.2f} W")
